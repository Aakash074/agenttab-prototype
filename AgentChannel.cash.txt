pragma cashscript ^0.14.0;

// Unidirectional micropayment channel for AI agents.
// The agent (owner) locks a budget here. For every API call it signs an off-chain
// voucher for the CUMULATIVE amount owed. The provider can settle at any time with
// the latest voucher: it receives exactly that amount and the remainder is forced
// back to the owner. After `timeout` the owner can reclaim everything.
contract AgentChannel(pubkey owner, pubkey provider, int timeout) {
    function settle(sig providerSig, datasig ownerVoucher, bytes8 amountBytes) {
        require(checkSig(providerSig, provider));

        // Voucher is bound to this exact channel (contract bytecode) -> no cross-channel replay.
        bytes msg = sha256(this.activeBytecode) + amountBytes;
        require(checkDataSig(ownerVoucher, msg, owner));

        int amount = int(amountBytes);
        int budget = tx.inputs[this.activeInputIndex].value;
        require(amount >= 546);
        require(amount <= budget - 2000);

        require(tx.outputs[0].lockingBytecode == new LockingBytecodeP2PKH(hash160(provider)));
        require(tx.outputs[0].value == amount);
        require(tx.outputs[1].lockingBytecode == new LockingBytecodeP2PKH(hash160(owner)));
        require(tx.outputs[1].value >= budget - amount - 1000);
    }

    function refund(sig ownerSig) {
        require(checkSig(ownerSig, owner));
        require(tx.time >= timeout);
    }
}
